\section{Topological spaces}
\begin{description}
    \item[topology] A set $T$ of subsets of $X$ such that \begin{itemize}
        \item $X \in T$
        \item $\varnothing \in T$
        \item $T$ is closed under \emph{arbitrary unions} and \emph{finite intersections}
    \end{itemize}
    \item[basis] A collection $\mathcal{B}$ of subsets of $X$ such that \begin{itemize}
        \item $\forall x \in X,\exists B\in \mathcal{B}: x\in B$
        \item $\forall B_{1},B_{2} \in \mathcal{B},x \in B_{1}\cap B_{2}\implies \exists B_{3}\in\mathcal{B}:x \in B_3\subseteq B_{1}\cap B_{2}$
    \end{itemize}
    The set $T_\mathcal{B}=\left\{ \bigcup \mathcal{F} \mid \mathcal{F} \subset \mathcal{B} \right\}$ is the topology generated by $\mathcal{B}$.

    \item[fineness] $T\text{ is finer than }S \iff S \subset T$
    \item[subspace] $(Y,T_Y)$ where $Y\subset X$ and $T_Y = \left\{ Y \cap U \mid U \in T\right\}$
    \item[neighborhood] $U$ is a neighborhood of $x \iff$ U is an open set containing $x$
    \item[closure] $\overline{Y} := \mathrm{Cl}_X(Y) := \bigcap \left\{ U\mid \complement_{X}U\in T\land Y\subset U \right\}$ is the smallest closed set containing $Y$.
    \item[interior] $\mathrm{Int}(Y):=\bigcup \left\{ U\mid U\in T \land U\subset Y \right\}$ is the largest open set contained in $Y$
    \item[boundary] $\partial_X Y := \mathrm{Cl}(Y)\setminus \mathrm{Int}(Y)$ 
    \item[boundary point] point $x$ in subspace $Y$ is a \emph{boundary point} of Y $\iff x \in \partial_X Y$ 
    \item[limit point] point $x$ in subspace $Y$ is a \emph{limit point} of $Y$ $\iff \forall \text{open neighborhood $U$ of $x$},U\cap Y \ne \varnothing$. $Y'$ denotes the set of limit points of $Y$.
    \item[isolated point] point $x$ in subspace $Y$ is an \emph{isolated point} of $Y$ $\iff \{x\}$ is open in $Y$
    \item[limit of sequence] $x$ for a sequence $\left( x_{n} \right)_{n\in \mathbb{N}}$ of points in $X$ $\iff \forall U\in T,x \in U \implies \exists N\in \mathbb{N}:\forall n\geq N,  x_{n}\in U$. Equivalently though informally, there comes a point where the terms of the sequence remain within a neighborhood of $x$.
    \item[convergence] existence of the limit of a sequence
    \item[pointwise convergence] for a sequence of functions $\left( f_i \right)_{i\in \mathbb{N}}$, if $\left( f_i(x) \right)_{i\in \mathbb{N}}$ converges for all $x$ in the domain of $f_i$.
    \item[uniform convergence] $\forall\varepsilon>0, \exists N\in \mathbb{N}:\forall n > N, \forall x \in X, d\left( f_n(x),f(x) \right) < \varepsilon$. converges to $f$
    %\item[boundary] $\partial$
\end{description}
\subsection{Properties}
\begin{description}
    \item[connected] if $\nexists\text{ open sets } U,V : U \cup V = X \land U \cap V = \varnothing$, or equivalently $X$ and $\varnothing$ are the only clopen sets
    \item[path-connected] if $\forall p,q \in X, \exists\text{ continuous }\left( \phi:[0,1] \to X  \right): \phi(0)=p \wedge \phi(1)=q$. Implies but is not implied by connectedness.
    \item[connected components] the equivalence class with respect to the relation $x \sim y \iff x,y \in \text{connected subspace $Y$ of $X$}$
    \item[path-connected components] the equivalence class with respect to the relation $x \sim y \iff \exists\text{path $\phi$ between $x$ \& $y$}$
    \item[compact] given that a \emph{cover} of $Y \subseteq X$ is a collection $\mathcal{C}$ of subsets of $X$ such that $\bigcup \mathcal{C} \supseteq Y$, and an \emph{open cover} is a cover whose elements are open, $Y$ is \emph{compact} if every open cover of $Y$ includes a finite open cover of $Y$.
    \item[limit point compact] if every infinite subset of $X$ has a limit point. Implied by but does not imply compactness.
\end{description}
\subsubsection{Separation axioms}
    For a topological space $X$ in which \emph{singletons are closed}, $X$ is said to be
\begin{description}
    \item[regular] if every pair of disjoint point and closed set is separated by neighborhoods\begin{itemize}[label=$\implies$]
        \item subspaces of $X$ are also regular
        \item products of $X$ with other regular spaces are also regular
    \end{itemize}
    \item[normal] if every pair of disjoint closed sets is separated by neighborhoods\begin{itemize}[label=$\implies$]
        \item every metrizable space is normal
        \item every compact T\textsubscript{2} space is normal
    \end{itemize}
    \item[T\textsubscript{2}] if every pair of disjoint points is separated by neighborhoods \begin{itemize}[label=$\implies$]
        \item compact subsets of $X$ are closed
        \item limits of sequences of points in $X$ are unique
        \item subspaces of $X$ are also T\textsubscript{2}
        \item products of $X$ with other T\textsubscript{2} spaces are also T\textsubscript{2}
    \end{itemize}
\end{description}
\subsection{Special topologies}
\begin{description}
    \item[trivial] $T=\{\varnothing, X\}$
    \item[discrete] $T=\mathcal{P}(X)$
    \item[metric] for metric space $(X,d)$, the topology generated by $\left\{ B_d(p,r) \mid p \in X \land r \in \mathbb{R} \right\}$.
    \item[order] for $(X,<)$, the topology generated by the collection $\mathcal{B}$ of the following subsets  $\forall a,b \in X$.\begin{itemize}
        \item $(a,b)$
        \item $[\min(X),b)$
        \item $(a,\max(X)]$
    \end{itemize}
    \item[product] for a collection of topological spaces of the form $(X, T_X)$, the collection of their underlying sets $\mathcal{X}$, the topology generated by $\left\{ \prod_{i\in I} U_i \mid U \in T_i \right\}$.
    \item on $\mathbb{R}, \forall x,y \in \mathbb{R}$:
    \begin{description}
        \item[standard] the metric topology induced by $d(x,y)=\left| x - y \right|$
        \item[lower] generated by the basis of $[x,y)$
        \item[upper] generated by the basis of $(x,y]$
        \item[$k$-topology] generated by the basis of $(x,y) \setminus \left\{ \frac{1}{z} \mid z \in \mathbb{Z}_+  \right\}$
    \end{description}
\end{description}
\section{Maps}
\begin{description}
    \item[injective] for $f:A\to B$ if $\forall a,b \in A, f(a)=f(b)\implies a = b$
    \item[surjective] if $\forall b \in B, \exists a \in A : f(b)=a$
    \item[bijective] if both injective and surjective
    \item[continuous] a map $f:X \to Y$ between topological spaces $(X,T)$ and $(Y,S)$ if $\forall V \in S, f^{-1}(V)\in T$. Continuous maps preserve \begin{itemize}
        \item connectedness
        \item path-connectedness
        \item compactness
    \end{itemize}
    \item[homeomorphism] a bijection $f:X\to Y$ if both $f$ and $f^{-1}$ are continuous
    \item[closed map] $f:X\to Y$ if $\forall A \subset X, A\text{ closed} \implies f(A)\text{ closed}$
\end{description}
\section{Metric spaces}
\begin{description}
    \item[metric] on a set $X$, the function $d:X \times X \to \mathbb{R}$ such that, $\forall x,y,z\in X,$ \begin{itemize}
        \item $d(x,y) \geq 0$
        \item $d(x,y) = 0 \iff x = y$
        \item $d(x,y)=d(y,x)$
        \item $d(x,z) \leq d(x,y)+d(y,z)$
    \end{itemize}
    \item[open ball] $B_d(p,r) := \left\{ x \mid d(x,p) < r  \right\}$
    \item A metric \emph{induces} the topology generated by the basis of all open balls using $d$.
    \item For $A \subset X$, \begin{itemize}
        \item $d(x,A):= \mathrm{inf}\left\{ d(x,a) \mid a \in A\right\}$
        \item $\mathrm{diam}\,A := \mathrm{sup}\left\{ d(a_{1},a_{2}) \mid a_{1},a_{2} \in A \right\}$
    \end{itemize}
    \item[uniformly continuous] a map $f:X\to Y$ between metric spaces if $\forall\varepsilon > 0, \exists\delta > 0: \forall x_0, x_1 \in X, d_X (x_0, x_1) < \delta \implies d_Y(f(x_0),f(x_1))<\varepsilon$
    \item[$\varepsilon$-neighborhood] the set $$\begin{aligned}
        U(A,\varepsilon) :\!\!&= \left\{ x \mid d(x,A) < \varepsilon \right\}\\
        &= \bigcup_{a \in A} B_d(a,\varepsilon)
    \end{aligned}$$
    \item[square metric] $d(p,q) = \max_i\left| p_i - q_i \right|$
\end{description}   
\section{Theorems}
\begin{description}
    \item[separation/subspace] If $X$ is separated by $C$ and $D$ and $Y$ is a connected subspace of $X$, then $Y$ lies entirely within either $C$ or $D$.
    \item[closure] $\mathrm{Cl}(Y)=Y\cup Y'$
    \item[closed/limit points] $Y$ is closed $\iff Y \subseteq Y'$
    \item[finite product connectedness] A finite product of connected spaces is connected
    \item[Uniform limit] For a sequence of functions $\left( f_n \right)$ from a topological space to a metric space, uniform convergence to $f \implies f$ is continuous.
    \item[Intermediate value] For continuous map $f:X\to Y$ from connected space $X$ to ordered set $Y$ in the order topology, $a,b\in X \land r\in Y \land f(a) < r < f(b) \implies \exists c\in X : f(c)=r$
    \item[Extreme value] For continuous map $f:X\to Y$ where $X$ is compact and $Y$ is an ordered set in the order topology, then $\forall x\in X, \exists c,d\in X : f(c) \leq f(x) \leq f(d)$
    \item[Lebesgue covering] For a compact metric space $X$ with open cover $\mathcal{C}$, there is some number $\delta > 0$ such that every subset of $X$ with diameter less than $\delta$ is contained in some member of $\mathcal{C}$.
    \item[Uniform continuity] $f:X\to Y$ is a continuous map from the compact metric space $\left( X, d_X \right)$ to the metric space $\left( Y, d_Y \right) \implies f$ is uniformly continuous.

    \emph{Pf.} $\mathcal{B}:=\left\{B_{d_Y}( y, \varepsilon/2) \mid y \in Y  \right\}$ for some $\varepsilon>0$ is an open covering of $Y$. Then $\mathcal{A} := \left\{ f^{-1}(B) \mid B \in \mathcal{B} \right\}$ is an open covering of $X$, with Lebesgue number $\delta$. Then $$\begin{aligned}
        \left( x_1, x_2 \in X \right) &\land \left( d_X(x_1,x_2)<\delta \right) \\
        &\implies \mathrm{diam}\left\{ x_1,x_2 \right\}<\delta \\
        &\implies \left\{ f(x_1), f(x_2)\right\} \in B \in \mathcal{B} \\
        &\implies d_Y(f(x_1),f(x_2))<\varepsilon
    \end{aligned}$$
    \item[l.u.b./compactness] For totally ordered set $X$ with the least upper bound property, closed intervals are compact in the order topology.
    \item[compact T\textsubscript{2}/uncountable] A nonempty compact T\textsubscript{2} space without isolated points is uncountable.
\end{description}
\section{Miscellanea}
\begin{description}
    \item[total order] a relation $<$ on a set $S$ such that $\forall a,b,c \in S$: \begin{itemize}
        \item $a < a$
        \item $a < b \land b < c \implies a < c$
        \item $a < b \land b < a \implies a = b$
    \end{itemize}
    \item[linear continuum] a totally ordered set $L$ such that: \begin{itemize}
        \item every subset bounded above has a least upper bound (l.u.b. property)
        \item $\forall a,b \in L, a < b \implies \exists c \in L : a < c < b$ (density)
    \end{itemize}
    \item[path] between points $p$ and $q$ in $X$, a continuous map $\phi:[0,1]\to X$ such that $\phi(0) = p$ and $\phi(1)=q$
\end{description}
