\section{Vector spaces}
A set $V$ with field $F$ and operations $+$ and $\cdot$ such that $\forall u,v,w \in V, \forall a,b \in F$:
    \begin{itemize}
        \item $u + v \in V$
        \item $a \cdot u \in V$
        \item $u + v = v + u$
        \item $(u + v) + w = u + (v + w)$
        \item $\exists 0 \in V : u + 0 = u$
        \item $\exists (-u) \in V : u + (-u) = 0$
        \item $1 \cdot u = u$
        \item $(ab) \cdot u = a \cdot (b \cdot u)$
        \item $(a + b) \cdot u = a \cdot u + b \cdot u$
        \item $a \cdot (u + v) = a \cdot u + a \cdot v$
    \end{itemize}
\begin{description}
    \item[subspace] a subset $S$ of $V$ that is itself a vector space under the same operations. It is sufficient that $S\neq\varnothing$ and $\forall u,v \in S\:\forall a,b \in F, au + bv \in S$.
    \item[linear combination] for a subset $S$ of $V$, a sum of scalar multiples of vectors in $S$, $\sum_{v\in S}a_v v$
    \item[linear independence] the property of a subset $S$ of $V$ where $\sum_{v\in S} a_v v = 0 \implies \forall v \in S, a_v = 0$
    \item[span] $\operatorname{Span}(S):= \left\{ \sum_{v \in S}a_v v \,\middle|\, \forall s \in S, a_s \in F \right\}$, the set of all linear combinations of vectors in $S$.
    \item[basis] a subset $B$ of $V$ that is linearly independent and spans $V$.
    \item[dimension] $\operatorname{dim}(V)=|B|$ for basis $B$.
    \item[sum of vector spaces] for subspaces $U$ and $W$ of vector space $V$, $U+W := \left\{ u + w \,\middle|\, u \in U \land w \in W \right\}$
\end{description}
\subsection{Theorems}
\begin{itemize}
    \item Zero vectors are unique for each vector space.
    \item Each vector has a unique additive inverse.
    \item Intersections of subspaces are subspaces.
    \item Spans are subspaces.
    \item Every vector space has a basis.
    \item Every vector has a unique representation w.r.t. a basis.
    \item Every basis of a vector space has the same cardinality.
    \item for a subset $S$ of $V$, $|S| > \dim(V) \implies S \text{ is linearly dependent}$.
    \item for a subset $S$ of $V$, the following imply $S$ is a basis:\begin{itemize}
        \item $S$ is linearly independent and spans $V$
        \item $S$ is linearly independent with $|S| = \dim(V)$ 
        \item $S$ spans $V$ with $|S| = \dim(V)$
    \end{itemize}
    \item for subspaces $W_1$ and $W_2$, $\dim(W_1 + W_2)=\dim(W_1) + \dim(W_2) - \dim(W_1 \cap W_2)$

\end{itemize}
\subsection{Standard bases}
\begin{description}
    \item[$F^n$] $\left\{ \left( \delta_{ij} \right)_{1\leq i \leq n} \,\middle|\, 1 \leq j \leq n\right\}$, the set of tuples where one term is $1_F$ and the rest are $0_F$.
    \item[$M_{m\times n}(F)$] $\left\{ E_{ij} \,\middle|\, 1\leq i \leq m \land 1 \leq j \leq n \right\}$, the set of matrices where one entry is $1_F$ and the rest are $0_F$.
    \item[$P_n(F)$] $\left\{ x \mapsto x^i \,\middle|\, 1 \leq i \leq n \right\}$
    \item[square symmetric $n\times n$ matrices] $\left\{ E_{ii} \,\middle|\, 1\leq i \leq n \right\} \cup \left\{ E_{ij} + E_{ji} \,\middle|\, 1\leq i < j \leq n \right\}$
\end{description}

\section{Linear maps}
For vector spaces $V$ and $W$ over field $F$, a map $f:V\to W$ such that $\forall a,b\in F\:\forall u,v \in V = f(au + bv) = af(u) + bf(v)$.
\begin{description}
    \item[kernel] $\operatorname{Ker}(f) := \left\{ v \,\middle|\, v\in V \land f(v) = 0 \right\}$ is the set of vectors in $V$ that $f$ takes to the zero vector in $W$. Also called the null space or notated $\operatorname{N}(f)$.
    \item[nullity] $\operatorname{null}(f) := \dim(\operatorname{Ker}(f))$
    \item[image]  $\operatorname{Im}(f) := \left\{ w \,\middle|\, w \in W \land \exists v\in V : w = f(v) \right\}$ is the set of outputs of $f$ in $W$. Also called the range or notated $\operatorname{R}(f)$.
    \item[rank] $\operatorname{rank}(f):=\dim(\operatorname{Im}(f))$
    \item[matrix of a linear map] If $B:=\left\{ b_i \,\middle|\, 1\leq i \leq \dim(V) \right\}$ is an ordered basis of $V$ and $C:=\left\{ c_i \,\middle|\, 1\leq i \leq \dim(W) \right\}$ an ordered basis of $W$, then there exists the matrix $[f]_B^C = \left[ f\left(b_1\right) \cdots f\left(b_{\dim(V)}\right)  \right]$ such that $f(v)=[f]_B^C v$.
    \item[coordinate vector] $\left[ v \right]_B := \left[ v_1 \cdots v_{\dim(V)} \right]^\intercal$, given that $v=\sum_{i=1}^{\dim(V)}v_i b_i$
\end{description}
\subsection{Theorems}
\begin{itemize}
    \item $f(0_V) = 0_W$
    \item $f\text{ is injective}\iff \operatorname{Ker}(f) = \left\{ 0 \right\} \iff \operatorname{null}(f)=0$
    \item $f\text{ is surjective}\iff \operatorname{Im}(f)=W \iff \operatorname{rank}(f)=\dim(f)$
    \item The inverse of an isomorphism is an isomorphism.
    \item Vector spaces are isomorphic iff they have the same dimension.
    \item All linear maps preserve linear dependence.
    \item Injective linear maps preserve linear independence.
    \item If $V$ is finite dimensional, then $\operatorname{null}(f)+\operatorname{rank}(f)=\dim(V)$
    \item Rank of a linear map $f$ can be found by putting $[f]$ in RREF and counting the pivots.
    \item Nullity of a linear map $f$ can be found via the previous theorem and the rank-nullity theorem.
\end{itemize}

\section{Matrices}
\begin{description}
    \item[addition] same dimensions, entrywise
    \item[scalar multiplication] entrywise
    \item[matrix multiplication] if $A$ is an $m\times p$ matrix and $B$ is an $p\times n$ matrix, then the product $AB$ is an $m\times n$ matrix defined by $[AB]_{ij} = \sum_{k=1}^p a_{ik}b_{kj}$. In other words, $[AB]_{ij}$ is the dot product of the $i$th row of $A$ and the $j$th column of $B$.
    \item[inverse] for square matrix $A$, the square matrix $A^{-1}$ such that $AA^{-1} = A^{-1}A=I$.
    \item[transpose] $[A^\intercal]_{ij} := [A]_{ji}$ 
\end{description}
\subsection{Theorems}
\begin{itemize}
    \item A matrix is invertible iff the only solution to $Av = 0$ is $v=0$.
    \item A matrix is invertible iff the determinant is nonzero.
\end{itemize}
